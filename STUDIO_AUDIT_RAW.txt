const title=document.getElementById("title");
const category=document.getElementById("category");
const body=document.getElementById("body");

function slug(text){

return text
.toLowerCase()
.normalize("NFD")
.replace(/[\u0300-\u036f]/g,"")
.replace(/[^a-z0-9\s-]/g,"")
.trim()
.replace(/\s+/g,"-");

}

function build(){

const today=new Date().toISOString().split("T")[0];

const s=slug(title.value||body.value.slice(0,25)||"capture");

return{

filename:`${today}-${s}.md`,

md:`---
title: "${title.value}"
category: "${category.value}"
created: ${today}
status: "Inbox"
---

# Ide

${body.value}

## Catatan

-

## Bisa berkembang menjadi

- Artikel
- Buku
- Video
- Newsletter
`
};

}

document.getElementById("copy").onclick=async()=>{

const d=build();

await navigator.clipboard.writeText(d.md);

alert("Capture disalin.");

};

document.getElementById("download").onclick=()=>{

const d=build();

const blob=new Blob([d.md],{type:"text/markdown"});

const url=URL.createObjectURL(blob);

const a=document.createElement("a");

a.href=url;

a.download=d.filename;

a.click();

URL.revokeObjectURL(url);

};
</script>

<script is:inline>
const title=document.getElementById("title");
const category=document.getElementById("category");
const body=document.getElementById("body");

const slugEl=document.getElementById("slug");
const fileEl=document.getElementById("filename");
const excerptEl=document.getElementById("excerpt");
const readingEl=document.getElementById("reading");

const pTitle=document.getElementById("preview-title");
const pCat=document.getElementById("preview-category");
const pBody=document.getElementById("preview-body");

function slugify(text){

return text
.toLowerCase()
.normalize("NFD")
.replace(/[\u0300-\u036f]/g,"")
.replace(/[^a-z0-9\s-]/g,"")
.trim()
.replace(/\s+/g,"-");

}

function generateMarkdown(){

const today=new Date().toISOString().split("T")[0];

const slug=slugify(title.value||"artikel-baru");

const words=body.value.trim().split(/\s+/).filter(Boolean).length;

const reading=Math.max(1,Math.ceil(words/200));

const excerpt=body.value.replace(/\n+/g," ").trim().slice(0,140);

return{
slug,
reading,
excerpt,
md:`---
title: "${title.value}"
kategori: "${category.value}"
status: "Draft"
tanggal: ${today}
excerpt: "${excerpt}"
readingTime: ${reading}
featured: false
---

# ${title.value}

${body.value}
`
};

}

function update(){

const data=generateMarkdown();

slugEl.textContent=data.slug;

fileEl.textContent=`${data.slug}.md`;

excerptEl.textContent=data.excerpt||"-";

readingEl.textContent=`${data.reading} menit`;

pTitle.textContent=title.value||"Judul artikel";

pCat.textContent=category.value;

pBody.textContent=body.value||"Isi artikel akan muncul di sini.";

}

[title,category,body].forEach(el=>el.addEventListener("input",update));

update();

document.getElementById("copy").onclick=async()=>{

const data=generateMarkdown();

await navigator.clipboard.writeText(data.md);

alert("Markdown disalin.");

};

document.getElementById("download").onclick=()=>{

const data=generateMarkdown();

const blob=new Blob([data.md],{type:"text/markdown;charset=utf-8;"});

const url=URL.createObjectURL(blob);

const a=document.createElement("a");

a.href=url;

a.download=`${data.slug}.md`;

document.body.appendChild(a);

a.click();

document.body.removeChild(a);

URL.revokeObjectURL(url);

};
</script>

const overlay=document.getElementById("command-palette");
const input=document.getElementById("command-input");
const list=document.getElementById("command-list");
const items=[...list.querySelectorAll("a")];

let index=0;

function highlight(){

items.forEach((el,i)=>{

el.classList.toggle("active",i===index);

});

}

function openPalette(){

overlay.classList.remove("hidden");

input.value="";

index=0;

highlight();

input.focus();

}

function closePalette(){

overlay.classList.add("hidden");

}

document.addEventListener("keydown",e=>{

if((e.ctrlKey||e.metaKey)&&e.key.toLowerCase()==="k"){

e.preventDefault();

openPalette();

}

if(e.key==="Escape"){

closePalette();

}

if(overlay.classList.contains("hidden")) return;

if(e.key==="ArrowDown"){

e.preventDefault();

index=(index+1)%items.length;

highlight();

}

if(e.key==="ArrowUp"){

e.preventDefault();

index=(index-1+items.length)%items.length;

highlight();

}

if(e.key==="Enter"){

e.preventDefault();

items[index].click();

}

});

input.addEventListener("input",()=>{

const q=input.value.toLowerCase();

const visible=items.filter(el=>{

const ok=el.textContent.toLowerCase().includes(q);

el.style.display=ok?"flex":"none";

return ok;

});

if(visible.length){

index=items.indexOf(visible[0]);

highlight();

}

});

overlay.addEventListener("click",e=>{

if(e.target===overlay){

closePalette();

}

});
</script>

  if (fs.existsSync(booksRoot)) {
    const folders = fs.readdirSync(booksRoot);

const NODE_REGEX = /^[TOPK]-\d{3}$/;
const EDITION_REGEX = /^\d{3}$/;
const SLUG_REGEX = /^[a-z0-9]+(?:-[a-z0-9]+)*$/;

/* ---------- World ---------- */

export function isValidWorld(value: string): value is WorldId {

  return WORLD_IDS.includes(value as WorldId);

}

/* ---------- Node ---------- */

export function isValidNodeId(value: string): value is NodeId {

  return NODE_REGEX.test(value);

}

/* ---------- Edition ---------- */

export function isValidEditionId(value: string): value is EditionId {

  return EDITION_REGEX.test(value);

}

/* ---------- Slug ---------- */

export function isValidSlug(value: string): value is ArticleSlug {

  return SLUG_REGEX.test(value);

}

/* ---------- Reading Time ---------- */

export function isValidReadingTime(value: string): value is ReadingTime {

  return /^\d+ menit membaca$/.test(value);

}

export function createSlug(input: string): ArticleSlug {

  const slug = input
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g,"")
    .replace(/[^a-z0-9]+/g,"-")
    .replace(/^-+|-+$/g,"")
    .replace(/-{2,}/g,"-");

  if(!isValidSlug(slug)){

    throw new Error(`Invalid slug: "${input}"`);

  }

  return slug;

}

export function createNodeId(prefix:"T"|"O"|"P"|"K",number:number):NodeId{

  const id=`${prefix}-${String(number).padStart(3,"0")}`;

  if(!isValidNodeId(id)){

    throw new Error(`Invalid NodeId: ${id}`);

  }

  return id;

}

export function createEditionId(number:number):EditionId{

  const id=String(number).padStart(3,"0");

  if(!isValidEditionId(id)){

    throw new Error(`Invalid EditionId: ${id}`);

  }

  return id;

}

   Assertions
===================================================== */

export function assertArticleMeta(article:ArticleMeta){

  if(!isValidSlug(article.slug))
    throw new Error(`Invalid slug: ${article.slug}`);

  if(!isValidWorld(article.world))
    throw new Error(`Invalid world: ${article.world}`);

  if(!isValidNodeId(article.node))
    throw new Error(`Invalid node: ${article.node}`);

  if(article.edition && !isValidEditionId(article.edition))
    throw new Error(`Invalid edition: ${article.edition}`);

  if(!isValidNodeId(article.archive))
    throw new Error(`Invalid archive: ${article.archive}`);

  if(!isValidReadingTime(article.readingTime))
    throw new Error(`Invalid reading time: ${article.readingTime}`);

  return article;
}

import http from "node:http";
import fs from "node:fs";
import { CONFIG } from "./config.mjs";
import { state } from "./state.mjs";

const json = (res, data, status = 200) => {
  res.writeHead(status, {
    "Content-Type": "application/json",
    "Access-Control-Allow-Origin": "http://localhost:4321",
    "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type"
  });

  res.end(JSON.stringify(data, null, 2));
};

const server = http.createServer((req, res) => {

  if (req.method === "OPTIONS") {
    res.writeHead(204);
    return res.end();
  }

  if (req.method === "GET" && req.url === "/health") {

    return json(res, {
      agent: "RAKAN Local Publish Agent",
      version: CONFIG.agentVersion,
      status: "ready"
    });

  }

  if (req.method === "GET" && req.url === "/status") {

    return json(res, {
      ...state,
      repository: CONFIG.repository,
      repositoryExists: fs.existsSync(CONFIG.repository)
    });

  }

  json(res, {
    error: "NOT_FOUND"
  }, 404);

});

server.listen(CONFIG.port, CONFIG.host, () => {

  console.log("");
  console.log("RAKAN Local Publish Agent");
  console.log("-------------------------");
  console.log(`Repository : ${CONFIG.repository}`);
  console.log(`Listening  : http://${CONFIG.host}:${CONFIG.port}`);
  console.log("");

});
